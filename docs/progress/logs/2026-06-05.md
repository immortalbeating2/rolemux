$workdir = Join-Path $env:TEMP ('rolemux-phase2-' + [guid]::NewGuid().ToString('N')); New-Item -ItemType Directory -Force -Path $workdir | Out-Null; $manifest = Join-Path $workdir 'rolemux-tasks.json'; @'
{
  "version": 1,
  "parentTask": { "title": "Phase 2 CLI dispatch" },
  "subtasks": [
    { "id": "one", "title": "One", "task": "Run mock provider.", "writePolicy": "readonly" }
  ]
}
'@ | Set-Content -LiteralPath $manifest -Encoding UTF8; $env:ROLEMUX_PROVIDER_CODEX_COMMAND = (Get-Command node).Source; $env:ROLEMUX_PROVIDER_CODEX_ARGS_PREFIX = (Resolve-Path tests\fixtures\mock-provider.mjs).Path; node .\dist\cli.js dispatch --manifest $manifest --providers codex:1 --workdir $workdir; Get-ChildItem -Recurse -LiteralPath (Join-Path $workdir '.rolemux\tasks') -File | Select-Object -ExpandProperty FullName; Remove-Item Env:\ROLEMUX_PROVIDER_CODEX_COMMAND -ErrorAction SilentlyContinue; Remove-Item Env:\ROLEMUX_PROVIDER_CODEX_ARGS_PREFIX -ErrorAction SilentlyContinue; Remove-Item -LiteralPath $workdir -Recurse -Force

- 下一阶段建议优先实现 worktree 创建、diff/patch 收集和 `merge --dry-run` 真实预览。
