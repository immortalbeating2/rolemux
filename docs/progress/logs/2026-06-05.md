```powershell
npx vitest run tests/core/git-worktree.test.ts tests/core/dispatch-artifacts.test.ts tests/commands/task-dispatch.test.ts
npm run typecheck
npm test
npm run build
$workdir = Join-Path $env:TEMP ('rolemux-phase3-' + [guid]::NewGuid().ToString('N')); New-Item -ItemType Directory -Force -Path $workdir | Out-Null; try { git -C $workdir init | Out-Null; git -C $workdir config user.email rolemux@example.invalid; git -C $workdir config user.name 'RoleMux Test'; Set-Content -LiteralPath (Join-Path $workdir 'README.md') -Value 'baseline' -Encoding UTF8; git -C $workdir add README.md; git -C $workdir commit -m baseline | Out-Null; $manifest = Join-Path $workdir 'rolemux-tasks.json'; @'
{
  "version": 1,
  "parentTask": { "title": "Phase 3 CLI dispatch" },
  "subtasks": [
    { "id": "write-code", "title": "Write code", "task": "Write one file.", "writePolicy": "isolated" }
  ]
}
'@ | Set-Content -LiteralPath $manifest -Encoding UTF8; $env:ROLEMUX_PROVIDER_CODEX_COMMAND = (Get-Command node).Source; $env:ROLEMUX_PROVIDER_CODEX_ARGS_PREFIX = (Resolve-Path tests\fixtures\write-file-provider.mjs).Path; node .\dist\cli.js dispatch --manifest $manifest --providers codex:1 --workdir $workdir | Out-String; $diff = Get-ChildItem -LiteralPath (Join-Path $workdir '.rolemux\tasks') -Recurse -Filter diff.patch | Select-Object -First 1; $worktree = Get-ChildItem -LiteralPath (Join-Path $workdir '.rolemux\tasks') -Recurse -Filter worktree.txt | Select-Object -First 1; if ($null -eq $diff -or $null -eq $worktree) { throw 'Missing Phase 3 artifacts' }; Get-Content -LiteralPath $diff.FullName | Select-String -Pattern 'worker-output.txt','created by isolated worker'; Get-Content -LiteralPath $worktree.FullName } finally { Remove-Item Env:\ROLEMUX_PROVIDER_CODEX_COMMAND -ErrorAction SilentlyContinue; Remove-Item Env:\ROLEMUX_PROVIDER_CODEX_ARGS_PREFIX -ErrorAction SilentlyContinue; $resolved = Resolve-Path -LiteralPath $workdir -ErrorAction SilentlyContinue; if ($resolved -and $resolved.Path.StartsWith([System.IO.Path]::GetTempPath(), [System.StringComparison]::OrdinalIgnoreCase)) { Remove-Item -LiteralPath $resolved.Path -Recurse -Force } }
git diff --check
```

- `git diff --check` 通过。
